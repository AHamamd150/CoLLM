import sys
import uproot
import awkward as ak
import vector
import numpy as np
import matplotlib.pyplot as plt

vector.register_awkward()

def find_branch(branches, candidates):
    for c in candidates:
        if c in branches:
            return c
    raise KeyError(f"None of {candidates} found in branches")

def main():
    if len(sys.argv) < 2:
        print("Usage: python analysis.py <input.root>")
        sys.exit(1)

    file = uproot.open(sys.argv[1])
    tree = file["Delphes"]
    branches = set(tree.keys())

    # Find branches
    e_pt_br = find_branch(branches, ["Electron.PT", "Electron/Electron.PT"])
    e_eta_br = find_branch(branches, ["Electron.Eta", "Electron/Electron.Eta"])
    mu_pt_br = find_branch(branches, ["Muon.PT", "Muon/Muon.PT"])
    mu_eta_br = find_branch(branches, ["Muon.Eta", "Muon/Muon.Eta"])
    j_pt_br = find_branch(branches, ["Jet.PT", "Jet/Jet.PT"])
    j_eta_br = find_branch(branches, ["Jet.Eta", "Jet/Jet.Eta"])
    j_btag_br = find_branch(branches, ["Jet.BTag", "Jet/Jet.BTag"])

    # Load arrays
    arrays = tree.arrays([e_pt_br, e_eta_br, mu_pt_br, mu_eta_br, 
                          j_pt_br, j_eta_br, j_btag_br], library="ak")

    electron_pt = arrays[e_pt_br]
    electron_eta = arrays[e_eta_br]
    muon_pt = arrays[mu_pt_br]
    muon_eta = arrays[mu_eta_br]
    jet_pt = arrays[j_pt_br]
    jet_eta = arrays[j_eta_br]
    jet_btag = arrays[j_btag_br]

    # Total events BEFORE cuts
    n_total = len(electron_pt)
    print(f"Total events: {n_total}")

    # Electron cuts: pT > 25, |eta| < 2.5
    e_mask = (electron_pt > 25) & (np.abs(electron_eta) < 2.5)
    electron_pt = electron_pt[e_mask]
    electron_eta = electron_eta[e_mask]

    # At least 1 electron
    mask = ak.num(electron_pt) >= 1
    electron_pt = electron_pt[mask]
    electron_eta = electron_eta[mask]
    muon_pt = muon_pt[mask]
    muon_eta = muon_eta[mask]
    jet_pt = jet_pt[mask]
    jet_eta = jet_eta[mask]
    jet_btag = jet_btag[mask]
    n_after_electron = len(electron_pt)
    print(f"After electron selection: {n_after_electron}")

    # At least 1 muon
    mask = ak.num(muon_pt) >= 1
    electron_pt = electron_pt[mask]
    electron_eta = electron_eta[mask]
    muon_pt = muon_pt[mask]
    muon_eta = muon_eta[mask]
    jet_pt = jet_pt[mask]
    jet_eta = jet_eta[mask]
    jet_btag = jet_btag[mask]
    n_after_muon = len(muon_pt)
    print(f"After muon selection: {n_after_muon}")

    # At least 2 jets
    mask = ak.num(jet_pt) >= 2
    electron_pt = electron_pt[mask]
    electron_eta = electron_eta[mask]
    muon_pt = muon_pt[mask]
    muon_eta = muon_eta[mask]
    jet_pt = jet_pt[mask]
    jet_eta = jet_eta[mask]
    jet_btag = jet_btag[mask]
    n_after_jet = len(jet_pt)
    print(f"After jet selection: {n_after_jet}")

    # At least 1 b-tagged jet
    n_bjets = ak.sum(jet_btag == 1, axis=1)
    mask = n_bjets >= 1
    electron_pt = electron_pt[mask]
    electron_eta = electron_eta[mask]
    muon_pt = muon_pt[mask]
    muon_eta = muon_eta[mask]
    jet_pt = jet_pt[mask]
    jet_eta = jet_eta[mask]
    jet_btag = jet_btag[mask]
    n_after_btag = len(jet_pt)
    print(f"After b-tag selection: {n_after_btag}")

    # Plot electron pT
    plt.figure(figsize=(8, 6))
    plt.hist(ak.to_numpy(ak.flatten(electron_pt)), bins=40, range=(0, 200), histtype='step')
    plt.xlabel("Electron pT [GeV]")
    plt.ylabel("Electrons")
    plt.title("Electron pT")
    plt.savefig("electron_pt.png", dpi=150)
    plt.close()

    # Plot muon pT
    plt.figure(figsize=(8, 6))
    plt.hist(ak.to_numpy(ak.flatten(muon_pt)), bins=40, range=(0, 200), histtype='step')
    plt.xlabel("Muon pT [GeV]")
    plt.ylabel("Muons")
    plt.title("Muon pT")
    plt.savefig("muon_pt.png", dpi=150)
    plt.close()

    # Plot jet pT
    plt.figure(figsize=(8, 6))
    plt.hist(ak.to_numpy(ak.flatten(jet_pt)), bins=40, range=(0, 200), histtype='step')
    plt.xlabel("Jet pT [GeV]")
    plt.ylabel("Jets")
    plt.title("Jet pT")
    plt.savefig("jet_pt.png", dpi=150)
    plt.close()

    # Plot jet b-tag
    plt.figure(figsize=(8, 6))
    plt.hist(ak.to_numpy(ak.flatten(jet_btag)), bins=40, range=(0, 1), histtype='step')
    plt.xlabel("Jet BTag")
    plt.ylabel("Jets")
    plt.title("Jet BTag")
    plt.savefig("jet_btag.png", dpi=150)
    plt.close()

if __name__ == "__main__":
    main()
